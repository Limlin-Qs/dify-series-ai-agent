### 2. C1 槽位抽取与校验（Code，输入=query+会话历史，输出槽位JSON）

```python
def main(query: str, sample_text: str) -> dict:
    import re, json
    t = query or ''
    LINE = {'01':'1号线','02':'2号线','03':'3号线','04':'4号线',
            '06':'6号线','08':'8号线','11':'蓝谷快线','13':'西海岸快线'}
    # 图号→线路归类（线路代号=编号第2-3位，Z/T后）
    graphs = {}
    for m in re.finditer(r'《?([ZTY]\d{5})》?', t):
        g = m.group(1); line = LINE.get(g[1:3])
        if line and g not in graphs.setdefault(line, []): graphs[line].append(g)
    # 运营时间：从检索范本按图号反查（确定性复用历史数据）
    ops = {}
    for row in re.finditer(r'\|\s*([^|]+)\|\s*([^|]*)\|\s*[^|]*\|\s*《?([ZTY]\d{5})》?\s*\|\s*([0-9:]+-[0-9:次日]+)', sample_text or ''):
        ops.setdefault(row.group(3), row.group(4).strip())
    # 日期：兼容"2026年10月1日"与"10月1日"（年份从前文取）
    dates = re.findall(r'20\d{2}\s*年\s*\d{1,2}\s*月\s*\d{1,2}\s*日', t) or \
            [f"2026年{x}" for x in re.findall(r'(?<![年\d])\d{1,2}\s*月\s*\d{1,2}\s*日', t)]
    has_item = any(k in t for k in ('调整','切换','延时','压缩','恢复','取消','新增','延长','改图','换图'))
    risk_known = ('风险' in t)
    missing = []
    if not dates: missing.append('生效日期')
    if not graphs: missing.append('各线路执行图号')
    if not has_item: missing.append('调图项点描述')
    if not risk_known: missing.append('风险项点（有/无）')
    slots = {'dates': dates, 'graphs': graphs, 'ops_from_sample': ops,
             'risk': risk_known, 'ats': ('基线' in t) or ('留存' in t)}
    return {'slots': json.dumps(slots, ensure_ascii=False),
            'missing': json.dumps(missing, ensure_ascii=False),
            'slots_ok': len(missing) == 0}
```
（2026-10-13 本地冒烟测试已过：图号归类、日期兼容、运营时间反查44条均正确；初始版有两处bug已修——线路代号取 g[1:3] 而非 g[:2]、日期兼容无年份写法）
入参：query=开始节点query；sample_text=RAG重排器-1.result（范本全文，用于反查运营时间）

### 4. C2 确认单组装（Code，替代原LLM确认单——确认单是确定性文本）

```python
def main(missing: str) -> dict:
    import json
    m = json.loads(missing or '[]')
    FMT = {'生效日期':'如"2026年10月1日起至另有通知时止"',
           '各线路执行图号':'如"1号线《Z01149》、3号线《T03061》"（Z=常规图、T=节假日/特殊图，前2位线路代号01/02/03/04/06/08/11/13）',
           '调图项点描述':'如"节假日图切换、1号线延时30分钟"',
           '风险项点（有/无）':'无 / 有（列出线路及内容，决定三章/四章结构）'}
    if not m:
        return {'clarify_text': ''}
    lines = ['《编制信息确认单》', '',
             '生成运作命令还缺以下要素，请补齐后回复"确认"：', '']
    for i, x in enumerate(m, 1):
        lines.append(f'{i}. **{x}** —— {FMT.get(x, "")}')
    lines += ['', '已提供的信息将自动保留，无需重复。']
    return {'clarify_text': '\n'.join(lines)}
```

### 6. C3 Markdown确定性组装（Code——表格句式全部代码拼装）

```python
def main(slots: str, items_json: str, sample_text: str) -> dict:
    import re, json
    s = json.loads(slots or '{}')
    items = json.loads(items_json or '[]')
    graphs = s.get('graphs', {}); ops = s.get('ops_from_sample', {})
    dates = s.get('dates', [])
    start = dates[0].replace(' ','') if dates else '【待补】'
    out = [f'2026年{{N}}号运作命令'.replace('{N}','【待补】'),
           '## 一、运作时间段',
           f'{start}起至另有通知时止执行的列车运行图如下：', '',
           '| 线别 | 运作时间段 | 执行运行图 | 运营时间 | 备用车安排 |',
           '|---|---|---|---|---|']
    ORDER = ['1号线','2号线','3号线','4号线','6号线','8号线','蓝谷快线','西海岸快线']
    for line in ORDER:
        gs = graphs.get(line)
        if gs:
            g = '、'.join(f'《{x}》' for x in gs)
            out.append(f'| {line} | {start}起 | {g} | {ops.get(gs[0],"【待补】")} | 图定备车 |')
        else:
            out.append(f'| {line} | {start}起 | 【沿用现行图，请核实】 | 【待补】 | 图定备车 |')
    out += ['', '注：表格中加粗运行图为本次调整运行图。', '',
            '## 二、ATS系统中留存运行图',
            f'{start}起至另有通知时止，各线路ATS系统中仅留存下列运行图。其他运行图由自动化中心移出ATS系统并做好备份。',
            '', '【依据2025年33号命令基线输出；如有新增/移出请指出】【待补：基线表格——二期由基线知识库结构化后代码直拼】', '']
    if s.get('risk') and '无' not in json.dumps(s, ensure_ascii=False):
        out += ['## 三、调图风险', '【按槽位风险项点列表格：线路|乘客服务|行车组织|生产运作】', '']
        ch = '四、注意事项'
    else:
        ch = '三、注意事项'
    out.append(f'## {ch}')
    for i, it in enumerate(items, 1):
        out.append(f'（{"一二三四五六七八九十"[i-1] if i<=10 else i}）{it}')
    return {'md_text': '\n'.join(out)}
```

### 7. C4 终检与校对要点（Code——把 apifox 断言搬进流程内）

```python
def main(md_text: str) -> dict:
    import re
    t = md_text or ''; issues = []
    if not re.search(r'# 20\d{2} ?年.*运作命令', t): issues.append('标题不规范')
    if '关于' in t.split('\n')[0] or '各中心' in t or '落款' in t: issues.append('残留通知体要素')
    if '| 线别 | 运作时间段 | 执行运行图 | 运营时间 | 备用车安排 |' not in t: issues.append('一章表头不符')
    for line in ['1号线','2号线','3号线','4号线','6号线','8号线','蓝谷快线','西海岸快线']:
        if f'| {line} ' not in t: issues.append(f'缺{line}行')
    if re.search(r'《\s*》|《》', t): issues.append('存在空图号占位')
    for g in re.findall(r'《([ZTY]\d{5})》', t):
        if g[:2] not in ('01','02','03','04','06','08','11','13'): issues.append(f'图号线路代号异常:{g}')
    if re.search(r'〔20\d{2}〕|发文字号', t): issues.append('出现发文字号（一票否决项）')
    report = ('终检通过，未发现问题。' if not issues else
              '请人工重点校对：\n' + '\n'.join(f'- {x}' for x in issues))
    return {'check_report': report, 'check_pass': len(issues) == 0}
```
**终检失败不阻断**（第一阶段定位：生成+人工微调）——check_report 照常输出给用户看。

### 8. I2 Word导出开关？（Ifs）+ 环境变量（护网降级）

- 画布加**环境变量** `export_enabled`（string，默认 "true"；护网期间在画布环境变量栏改成 "false"，不动连线）
- I2 条件：`export_enabled = "true"`（string比较）。true→Word文档生成；false→直接→结束
- 降级输出已由结束节点承载：download_url 为空，对话流模板可加一行提示文本（见下）
- **原理备注**：MCP 挂了是断流、Ifs 事后救不了（2026-10-13 已实测）；所以降级必须是**事前开关**——在调 MCP 之前判断，而不是失败后兜底

## 四、与33份语料的关系（为什么够生成全量）

- 结构骨架：C3 模板化，来自33份统计（命令模板要素提炼.md）
- 运营时间/备用车：C1 从检索范本按图号确定性反查
- 注意事项套话：L1 从范本继承+新写，33份已覆盖全部固定套话
- 图号：用户槽位提供或基线预填，永不编造
- 人工只微调：【待补】项+check_report 要点——即"第一阶段"的工作量边界
