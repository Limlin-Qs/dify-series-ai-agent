# 天气查询助手 - 极视角节点代码与提示词

## 使用约定
- 代码块顶层只保留一个 `def main(...)`
- 所有import和自定义函数写在main内部
- 返回字典的key与输出变量一致
- 所有输出必须可JSON序列化
- 模型输出使用 `text:string`，直接输出Markdown

## N02: 解析输入 (代码节点)

### 输入变量
| 变量 | 类型 | 说明 |
|---|---|---|
| data | string | 原始JSON字符串 |

### 输出变量
| 变量 | 类型 | 说明 |
|---|---|---|
| city | string | 标准化城市名 |
| date | string | 查询日期（默认今天） |
| is_valid | boolean | 输入是否有效 |
| error_msg | string | 错误信息 |

### 代码

```python
def main(data: str) -> dict:
    import json
    from datetime import datetime, timedelta
    
    def parse_relative_date(date_str: str) -> str:
        """解析相对日期"""
        if not date_str or date_str.strip() == "":
            return datetime.now().strftime("%Y-%m-%d")
        
        date_str = date_str.strip()
        today = datetime.now()
        
        if date_str == "今天":
            return today.strftime("%Y-%m-%d")
        elif date_str == "明天":
            return (today + timedelta(days=1)).strftime("%Y-%m-%d")
        elif date_str == "后天":
            return (today + timedelta(days=2)).strftime("%Y-%m-%d")
        else:
            # 尝试解析具体日期格式
            for fmt in ["%Y-%m-%d", "%Y/%m/%d", "%m-%d", "%m/%d"]:
                try:
                    parsed = datetime.strptime(date_str, fmt)
                    # 如果只有月日，补充年份
                    if "%Y" not in fmt:
                        parsed = parsed.replace(year=today.year)
                    return parsed.strftime("%Y-%m-%d")
                except ValueError:
                    continue
            
            # 无法解析，返回今天
            return today.strftime("%Y-%m-%d")
    
    def normalize_city(city_name: str) -> str:
        """标准化城市名"""
        if not city_name:
            return ""
        
        city_name = city_name.strip()
        
        # 常见别名映射
        aliases = {
            "北京": "北京",
            "北京市": "北京",
            "北平": "北京",
            "上海": "上海",
            "上海市": "上海",
            "广州": "广州",
            "广州市": "广州",
            "深圳": "深圳",
            "深圳市": "深圳",
        }
        
        return aliases.get(city_name, city_name)
    
    def validate_city(city_name: str) -> tuple[bool, str]:
        """校验城市名有效性"""
        if not city_name or len(city_name.strip()) == 0:
            return False, "城市名称不能为空"
        
        if len(city_name) > 50:
            return False, "城市名称过长"
        
        # 基本字符校验（允许中文、英文、空格）
        import re
        if not re.match(r'^[\u4e00-\u9fa5a-zA-Z\s]+$', city_name):
            return False, "城市名称包含非法字符"
        
        return True, ""
    
    # 主逻辑
    try:
        input_data = json.loads(data)
    except json.JSONDecodeError:
        return {
            "city": "",
            "date": "",
            "is_valid": False,
            "error_msg": "输入数据格式错误，请提供有效的JSON格式"
        }
    
    city_raw = input_data.get("city", "")
    date_raw = input_data.get("date", "")
    
    # 标准化处理
    city = normalize_city(city_raw)
    date = parse_relative_date(date_raw)
    
    # 校验
    is_valid, error_msg = validate_city(city)
    
    return {
        "city": city,
        "date": date,
        "is_valid": is_valid,
        "error_msg": error_msg if not is_valid else ""
    }
```

### 代码

```python
def main(raw_weather: dict, is_error: int) -> dict:
    def safe_get(data: dict, keys: list, default=None):
        """安全获取嵌套字典值"""
        result = data
        for key in keys:
            if isinstance(result, dict):
                result = result.get(key)
                if result is None:
                    return default
            else:
                return default
        return result
    
    def get_condition_level(condition: str) -> str:
        """根据天气状况返回等级描述"""
        if not condition:
            return "未知"
            
        condition_lower = condition.lower()
        
        sunny = ["晴", "sunny", "clear"]
        cloudy = ["多云", "cloudy", "partly"]
        overcast = ["阴", "overcast"]
        rain = ["雨", "rain", "shower"]
        snow = ["雪", "snow"]
        storm = ["暴", "storm", "thunder"]
        
        for keywords in [sunny, cloudy, overcast, rain, snow, storm]:
            if any(kw in condition_lower for kw in keywords):
                return keywords[0]
        
        return condition
    
    def get_uv_level(uv_index: float) -> str:
        """根据UV指数返回等级"""
        if uv_index is None:
            return "未知"
        if uv_index <= 2:
            return "最弱"
        elif uv_index <= 5:
            return "弱"
        elif uv_index <= 7:
            return "中等"
        elif uv_index <= 10:
            return "强"
        else:
            return "极强"
    
    def get_aqi_level(aqi: int) -> str:
        """根据AQI返回空气质量等级"""
        if aqi is None:
            return "未知"
        if aqi <= 50:
            return "优"
        elif aqi <= 100:
            return "良"
        elif aqi <= 150:
            return "轻度污染"
        elif aqi <= 200:
            return "中度污染"
        elif aqi <= 300:
            return "重度污染"
        else:
            return "严重污染"
    
    def generate_tips(weather: dict) -> list:
        """根据天气生成生活建议"""
        tips = []
        
        temp = weather.get("temperature", {}).get("current", 25)
        condition = weather.get("condition", "")
        humidity = weather.get("humidity", 60)
        wind_speed = weather.get("wind_speed", 0)
        uv = weather.get("uv_index", {})
        uv_value = uv.get("value") if isinstance(uv, dict) else uv
        
        # 温度建议
        if temp >= 35:
            tips.append("高温预警：避免长时间户外活动，注意防暑降温")
        elif temp >= 30:
            tips.append("天气炎热：注意补水，做好防晒措施")
        elif temp <= 5:
            tips.append("严寒天气：注意保暖，预防感冒")
        elif temp <= 10:
            tips.append("气温较低：适当增添衣物")
        
        # 天气状况建议
        if any(kw in condition for kw in ["雨", "rain"]):
            tips.append("有降雨：出行请携带雨具，注意路滑")
        if any(kw in condition for kw in ["雪", "snow"]):
            tips.append("有降雪：路面可能结冰，减速慢行")
        
        # 湿度建议
        if humidity >= 80:
            tips.append("湿度较高：体感闷热，注意通风")
        elif humidity <= 30:
            tips.append("空气干燥：注意保湿和防火")
        
        # 风力建议
        if wind_speed >= 7:
            tips.append("大风天气：避免高空作业，固定易落物品")
        elif wind_speed >= 5:
            tips.append("风力较大：户外活动注意防风")
        
        # 紫外线建议
        if isinstance(uv_value, (int, float)) and uv_value >= 8:
            tips.append("紫外线很强：外出务必涂抹防晒霜，佩戴遮阳帽")
        elif isinstance(uv_value, (int, float)) and uv_value >= 3:
            tips.append("紫外线较强：建议做基本防晒措施")
        
        # 默认提示
        if not tips:
            tips.append("天气适宜，适合各类户外活动")
        
        return tips[:6]  # 最多6条建议
    
    # 检查是否有错误或无数据
    if is_error == 1 or not raw_weather:
        return {
            "weather_info": {},
            "has_data": False
        }
    
    # 提取并标准化数据
    try:
        city = safe_get(raw_weather, ["city", "name"], safe_get(raw_weather, ["city"], ""))
        date = safe_get(raw_weather, ["date"], safe_get(raw_weather, ["forecast_date"], ""))
        
        # 温度信息
        temp_current = safe_get(raw_weather, ["temperature", "current"], 
                               safe_get(raw_weather, ["temp"], 25))
        temp_min = safe_get(raw_weather, ["temperature", "min"],
                           safe_get(raw_weather, ["temp_min"], temp_current))
        temp_max = safe_get(raw_weather, ["temperature", "max"],
                           safe_get(raw_weather, ["temp_max"], temp_current))
        
        # 天气状况
        condition = safe_get(raw_weather, ["condition", "text"],
                            safe_get(raw_weather, ["weather", "description"], "晴"))
        condition_level = get_condition_level(condition)
        
        # 其他指标
        humidity = safe_get(raw_weather, ["humidity"], 60)
        wind_speed = safe_get(raw_weather, ["wind", "speed"],
                             safe_get(raw_weather, ["wind_speed"], 0))
        wind_direction = safe_get(raw_weather, ["wind", "direction"],
                                 safe_get(raw_weather, ["wind_direction"], "东风"))
        
        # 空气质量
        aqi = safe_get(raw_weather, ["air_quality", "aqi"],
                      safe_get(raw_weather, ["aqi"], None))
        air_quality = get_aqi_level(aqi)
        
        # 紫外线
        uv_raw = safe_get(raw_weather, ["uv_index"], safe_get(raw_weather, ["uv"], None))
        if isinstance(uv_raw, dict):
            uv_value = uv_raw.get("value", uv_raw.get("index", 0))
        else:
            uv_value = uv_raw if uv_raw else 0
        uv_level = get_uv_level(uv_value)
        
        # 日出日落
        sunrise = safe_get(raw_weather, ["sun", "rise"],
                          safe_get(raw_weather, ["sunrise"], "06:00"))
        sunset = safe_get(raw_weather, ["sun", "set"],
                         safe_get(raw_weather, ["sunset"], "18:30"))
        
        # 构建标准化结构
        weather_info = {
            "city": city,
            "date": date,
            "temperature": {
                "current": float(temp_current),
                "min": float(temp_min),
                "max": float(temp_max)
            },
            "condition": condition,
            "condition_level": condition_level,
            "humidity": int(humidity),
            "wind_speed": float(wind_speed),
            "wind_direction": str(wind_direction),
            "air_quality": air_quality,
            "aqi": int(aqi) if aqi else None,
            "uv_index": {
                "value": float(uv_value) if uv_value else 0,
                "level": uv_level
            },
            "sunrise": str(sunrise),
            "sunset": str(sunset)
        }
        
        # 生成生活建议
        tips = generate_tips(weather_info)
        weather_info["tips"] = tips
        
        return {
            "weather_info": weather_info,
            "has_data": True
        }
        
    except Exception as e:
        return {
            "weather_info": {},
            "has_data": False
        }
```

### 系统提示词

```
你是一位专业、友好的气象播报员。你的任务是根据提供的天气数据，为用户生成一份清晰、实用的天气预报报告。

## 角色定位
- 专业但不生硬，使用通俗易懂的语言
- 关注用户实际需求，提供实用建议
- 语气亲切自然，像朋友间的分享

## 输出要求
1. 必须使用以下固定的Markdown格式：
   - 一级标题：## {城市名}天气预报
   - 二级标题使用：### 温度概况、### 天气状况、### 生活指数、### 温馨提示
   
2. 报告内容必须包含：
   - 日期和城市信息
   - 当前温度、最高温、最低温
   - 天气状况描述（用2-3句话形象描述）
   - 主要气象指标（湿度、风向风力、空气质量、紫外线）
   - 根据数据给出穿衣、出行、运动建议
   - 特别提醒（如有极端天气）

3. 数据使用规则：
   - 所有数字必须来自输入数据，不得编造
   - 如果某项数据缺失，跳过该部分或说明"暂无数据"
   - 温度单位统一使用摄氏度（℃）

4. 语言风格：
   - 使用emoji增加可读性（☀️🌧️❄️💨等）
   - 段落简洁，每段不超过4行
   - 重点信息可以加粗显示

5. 输出长度：
   - 总字数控制在800-1200字
   - 不宜过短导致信息不足，也不宜过长影响阅读

## 禁止事项
- ❌ 不得编造任何不在输入中的数据
- ❌ 不得使用"据预测"、"可能"等不确定词汇描述已有数据
- ❌ 不得输出JSON格式，只输出纯Markdown文本
- ❌ 不得包含技术术语或API相关信息
- ❌ 不得省略必需的章节标题

## 失败处理
如果输入数据明显不完整或异常，在报告开头明确说明："⚠️ 数据异常，以下报告基于有限数据生成"，然后基于可用数据生成尽可能完整的内容。
```

## N06: 报告组装 (代码节点)

### 输入变量
| 变量 | 类型 | 说明 |
|---|---|---|

| weather_info | dict | 结构化天气数据 |

### 输出变量
| 变量 | 类型 | 说明 |
|---|---|---|
| md_text | string | 最终完整报告 |
| summary | string | 一句话摘要 |

### 代码

```python
def main(text: str, weather_info: dict) -> dict:
    from datetime import datetime
    
    def generate_summary(info: dict) -> str:
        """生成一句话摘要"""
        if not info:
            return "无法获取天气信息"
        
        city = info.get("city", "未知城市")
        date = info.get("date", "")
        temp = info.get("temperature", {})
        current_temp = temp.get("current", "--")
        condition = info.get("condition", "未知")
        
        summary = f"{city}({date})：{current_temp}℃，{condition}"
        return summary
    
    def add_header_footer(report_text: str, info: dict) -> str:
        """添加报告头尾信息"""
        now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        city = info.get("city", "") if info else "未知"
        date = info.get("date", "") if info else ""
        
        header = f"> 📅 报告生成时间：{now}\n"
        header += f"> 📍 查询城市：{city}\n"
        header += f"> 📆 查询日期：{date}\n\n"
        
        footer = "\n\n---\n*以上信息由天气查询助手自动生成，仅供参考*"
        
        return header + report_text + footer
    
    def validate_report(report: str) -> tuple[bool, str]:
        """校验报告完整性"""
        if not report or len(report.strip()) < 50:
            return False, "报告内容过短，可能生成失败"
        
        required_headers = ["## ", "### "]
        for header in required_headers:
            if header not in report:
                return False, f"缺少必要标题格式：{header}"
        
        return True, ""
    
    # 校验模型输出
    is_valid, error_msg = validate_report(text)
    
    if not is_valid:
        # 输出不完整，生成简化版报告
        summary = generate_summary(weather_info)
        fallback_report = f"## ⚠️ 天气信息\n\n{summary}\n\n详细报告生成异常：{error_msg}"
        md_text = add_header_footer(fallback_report, weather_info)
        return {
            "md_text": md_text,
            "summary": summary
        }
    
    # 正常组装
    summary = generate_summary(weather_info)
    md_text = add_header_footer(text, weather_info)
    
    return {
        "md_text": md_text,
        "summary": summary
    }
```

```python
def main(error_msg: str, is_error: int) -> dict:
    def generate_input_error(msg: str) -> dict:
        """生成输入错误提示"""
        md_text = f"""## ⚠️ 输入错误

抱歉，您的请求无法处理。

**错误原因：**{msg}

**解决建议：**
- 请检查城市名称是否正确
- 支持格式：`{{"city": "北京"}}`
- 可选参数：`{{"city": "上海", "date": "明天"}}`

**示例：**
- 查询北京今天：`{{"city": "北京"}}`

---
*如有疑问，请重新输入正确的查询条件*
"""
        return {
            "md_text": md_text,
            "summary": f"输入错误：{msg}"
        }
    
    def generate_api_error() -> dict:

抱歉，天气服务暂时无法响应。

**可能原因：**
- 天气服务器正在维护
- 网络连接不稳定
- 服务暂时过载

**您可以：**
- 稍后重试（建议1-2分钟后）
- 检查网络连接
- 联系管理员确认服务状态

---
*我们正在努力恢复服务，感谢您的耐心等待*
"""
        return {
            "md_text": md_text,
            "summary": "天气服务暂时不可用，请稍后重试"
        }
    
    # 判断错误类型
    if error_msg and len(error_msg.strip()) > 0:
        # 输入验证错误
        return generate_input_error(error_msg)
    elif is_error == 1:
        # API调用错误
        return generate_api_error()
    else:
        # 未知错误
        return generate_api_error()
```

## 注意事项

1. **MCP服务配置：** 在部署前需确保天气MCP服务已正确配置，包括服务ID、工具名和认证信息

2. **城市支持范围：** 当前支持中国主要城市和国际热门城市，如需扩展请在N02的validate_city函数中添加

3. **数据源依赖：** 天气数据的准确性和完整性取决于外部API，建议在生产环境中添加缓存机制

4. **性能优化：** 如需高并发支持，建议在N03前增加缓存检查节点，避免重复调用API

5. **国际化：** 如需支持多语言输出，可在N05的系统提示词中增加语言选项，或在N01开始节点增加language参数
